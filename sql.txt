DROP DATABASE IF EXISTS schooldb;
CREATE DATABASE schooldb;
USE schooldb;

CREATE TABLE Students (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    DA VARCHAR(50) NOT NULL,
    Nom VARCHAR(100) NOT NULL,
    Prenom VARCHAR(100) NOT NULL,
    Password VARCHAR(50) NOT NULL,
    AverageGrade DOUBLE NOT NULL,
    Role VARCHAR(50) NOT NULL
);

CREATE TABLE Tutors (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    Nom VARCHAR(100) NOT NULL,
    Prenom VARCHAR(100) NOT NULL,
    DA VARCHAR(50) NOT NULL,
    Password VARCHAR(50) NOT NULL,
    Subject VARCHAR(100) NOT NULL,
    Availability VARCHAR(100) NOT NULL,
    AverageGrade DOUBLE NOT NULL,
    IsValidated BIT NOT NULL,
    Role VARCHAR(50) NOT NULL,
    NumberOfRatings INT NOT NULL DEFAULT 0,
    TotalRatings INT NOT NULL DEFAULT 0
);

CREATE TABLE Appointments (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    StudentId INT NOT NULL,
    TutorId INT NOT NULL,
    DateRDV DATETIME NOT NULL,
    Status VARCHAR(50) DEFAULT 'Pending',
    FOREIGN KEY (StudentId) REFERENCES Students(Id),
    FOREIGN KEY (TutorId) REFERENCES Tutors(Id)
);

CREATE TABLE CompletedAppointments (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    AppointmentId INT NOT NULL,
    StudentId INT NOT NULL,
    TutorId INT NOT NULL,
    CompletedDate DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (AppointmentId) REFERENCES Appointments(Id),
    FOREIGN KEY (StudentId) REFERENCES Students(Id),
    FOREIGN KEY (TutorId) REFERENCES Tutors(Id),
    UNIQUE (AppointmentId)
);

CREATE TABLE Reviews (
    Id INT AUTO_INCREMENT PRIMARY KEY,
    CompletedAppointmentId INT NOT NULL,
    TutorId INT NOT NULL,
    StudentId INT NOT NULL,
    Comment VARCHAR(255),
    Rating INT NOT NULL,
    FOREIGN KEY (CompletedAppointmentId) REFERENCES CompletedAppointments(Id),
    FOREIGN KEY (TutorId) REFERENCES Tutors(Id),
    FOREIGN KEY (StudentId) REFERENCES Students(Id),
    CHECK (Rating >= 1 AND Rating <= 5),
    UNIQUE (CompletedAppointmentId, StudentId)
);
